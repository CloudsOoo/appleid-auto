# Apple ID 自动解锁系统 - 数据库设计

## 数据库选型
- **数据库**: PostgreSQL 15+
- **字符集**: UTF-8
- **时区**: UTC

---

## 表结构设计

### 1. users - 用户表
存储所有用户信息（包括普通用户和管理员）

```sql
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,           -- 用户名
    email VARCHAR(100) UNIQUE NOT NULL,             -- 邮箱
    password_hash VARCHAR(255) NOT NULL,            -- 密码哈希
    role VARCHAR(20) NOT NULL DEFAULT 'user',       -- 角色: user, admin
    is_active BOOLEAN DEFAULT TRUE,                 -- 是否激活
    is_verified BOOLEAN DEFAULT FALSE,              -- 邮箱是否验证

    -- 2FA 相关
    two_factor_enabled BOOLEAN DEFAULT FALSE,       -- 是否启用 2FA
    two_factor_secret VARCHAR(32),                  -- 2FA 密钥

    -- 个人信息
    full_name VARCHAR(100),                         -- 全名
    phone VARCHAR(20),                              -- 电话
    avatar_url VARCHAR(255),                        -- 头像

    -- 时间戳
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login_at TIMESTAMP,

    -- 索引
    INDEX idx_username (username),
    INDEX idx_email (email),
    INDEX idx_created_at (created_at)
);
```

---

### 2. packages - 套餐模板表
定义不同的权限套餐

```sql
CREATE TABLE packages (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,                     -- 套餐名称
    description TEXT,                               -- 套餐描述

    -- 权限配置
    max_accounts INTEGER DEFAULT 10,                -- 最大账号数
    max_share_pages INTEGER DEFAULT 5,              -- 最大分享页数
    max_nodes INTEGER DEFAULT 1,                    -- 最大节点数
    min_unlock_interval INTEGER DEFAULT 3600,       -- 最短解锁间隔(秒)

    -- 功能权限
    allow_custom_html BOOLEAN DEFAULT FALSE,        -- 允许自定义 HTML
    allow_view_password_history BOOLEAN DEFAULT FALSE, -- 允许查看历史密码
    allow_batch_import BOOLEAN DEFAULT TRUE,        -- 允许批量导入
    allow_api_access BOOLEAN DEFAULT FALSE,         -- 允许 API 访问

    -- 配额
    max_unlock_per_day INTEGER DEFAULT 100,         -- 每日最大解锁次数
    max_import_per_time INTEGER DEFAULT 100,        -- 单次导入限制

    -- 价格信息（仅供展示）
    price DECIMAL(10, 2),                           -- 价格
    currency VARCHAR(10) DEFAULT 'CNY',             -- 货币

    -- 状态
    is_active BOOLEAN DEFAULT TRUE,                 -- 是否启用
    sort_order INTEGER DEFAULT 0,                   -- 排序

    -- 时间戳
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_is_active (is_active)
);
```

### 3. cards - 卡密表
存储生成的所有卡密

```sql
CREATE TABLE cards (
    id SERIAL PRIMARY KEY,
    card_code VARCHAR(64) UNIQUE NOT NULL,          -- 卡密代码
    card_type VARCHAR(20) NOT NULL,                 -- 类型: time, usage, permanent

    -- 套餐关联
    package_id INTEGER REFERENCES packages(id),     -- 关联套餐

    -- 时长/次数配置
    duration_days INTEGER,                          -- 时长(天) - time类型
    usage_count INTEGER,                            -- 次数 - usage类型

    -- 状态
    status VARCHAR(20) DEFAULT 'unused',            -- 状态: unused, used, expired, revoked

    -- 激活信息
    activated_by INTEGER REFERENCES users(id),      -- 激活用户
    activated_at TIMESTAMP,                         -- 激活时间
    activated_ip VARCHAR(45),                       -- 激活 IP
    expires_at TIMESTAMP,                           -- 到期时间

    -- 使用统计
    used_count INTEGER DEFAULT 0,                   -- 已使用次数

    -- 批次信息
    batch_id VARCHAR(50),                           -- 批次 ID

    -- 备注
    note TEXT,                                      -- 备注

    -- 创建信息
    created_by INTEGER REFERENCES users(id),        -- 创建者
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_card_code (card_code),
    INDEX idx_status (status),
    INDEX idx_activated_by (activated_by),
    INDEX idx_batch_id (batch_id)
);
```

CREATE TABLE user_permissions (

### 6. apple_accounts - Apple ID 账号表
存储用户管理的 Apple ID 账号

```sql
CREATE TABLE apple_accounts (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id),           -- 所属用户

    -- 账号信息
    apple_id VARCHAR(100) NOT NULL,                 -- Apple ID (邮箱)
    current_password VARCHAR(255),                  -- 当前密码（加密）
    security_questions JSONB,                       -- 密保问题（加密）

    -- 账号状态
    status VARCHAR(50) DEFAULT 'normal',            -- 状态: normal, locked, processing, error
    lock_status BOOLEAN DEFAULT FALSE,              -- 是否锁定
    two_factor_enabled BOOLEAN DEFAULT FALSE,       -- 是否开启 2FA

    -- App Store 信息
    app_store_country VARCHAR(10),                  -- 地区代码
    app_store_language VARCHAR(10),                 -- 语言
    account_name VARCHAR(100),                      -- 账户姓名

    -- 设备信息
    devices_count INTEGER DEFAULT 0,                -- 设备数量
    has_lost_mode BOOLEAN DEFAULT FALSE,            -- 是否有丢失模式设备

    -- 分类标签
    tags JSONB,                                     -- 标签数组 ["tag1", "tag2"]
    category VARCHAR(50),                           -- 分类

    -- 自动化配置
    auto_unlock BOOLEAN DEFAULT TRUE,               -- 自动解锁
    auto_disable_2fa BOOLEAN DEFAULT FALSE,         -- 自动关闭 2FA
    auto_change_password BOOLEAN DEFAULT FALSE,     -- 自动修改密码
    auto_remove_devices BOOLEAN DEFAULT FALSE,      -- 自动删除设备

    -- 检测配置
    check_interval INTEGER DEFAULT 3600,            -- 检测间隔(秒)
    last_checked_at TIMESTAMP,                      -- 上次检测时间
    next_check_at TIMESTAMP,                        -- 下次检测时间

    -- 操作记录
    last_unlocked_at TIMESTAMP,                     -- 上次解锁时间
    unlock_count INTEGER DEFAULT 0,                 -- 解锁次数
    last_error TEXT,                                -- 最后错误信息
    error_count INTEGER DEFAULT 0,                  -- 错误次数

    -- 备注
    note TEXT,                                      -- 备注

    -- 时间戳
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_user_id (user_id),
    INDEX idx_apple_id (apple_id),
    INDEX idx_status (status),
    INDEX idx_lock_status (lock_status),
    INDEX idx_next_check_at (next_check_at),
    UNIQUE (user_id, apple_id)
);
```

### 8. unlock_tasks - 解锁任务表
记录所有解锁任务

```sql
CREATE TABLE unlock_tasks (
    id SERIAL PRIMARY KEY,
    account_id INTEGER REFERENCES apple_accounts(id),
    user_id INTEGER REFERENCES users(id),

    -- 任务信息
    task_type VARCHAR(50) NOT NULL,                 -- 类型: unlock, disable_2fa, change_password, remove_devices
    task_id VARCHAR(100) UNIQUE,                    -- Celery 任务 ID

    -- 任务状态
    status VARCHAR(50) DEFAULT 'pending',           -- 状态: pending, processing, success, failed, cancelled
    priority INTEGER DEFAULT 0,                     -- 优先级

    -- 执行信息
    node_id INTEGER,                                -- 执行节点 ID
    proxy_id INTEGER,                               -- 使用的代理 ID

    -- 结果
    result JSONB,                                   -- 执行结果（JSON）
    error_message TEXT,                             -- 错误信息
    retry_count INTEGER DEFAULT 0,                  -- 重试次数
    max_retries INTEGER DEFAULT 3,                  -- 最大重试次数

    -- 时间信息
    started_at TIMESTAMP,                           -- 开始时间
    completed_at TIMESTAMP,                         -- 完成时间
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_account_id (account_id),
    INDEX idx_user_id (user_id),
    INDEX idx_status (status),
    INDEX idx_task_id (task_id),
    INDEX idx_created_at (created_at)
);
```

## 数据库性能优化

### 1. 分区表（大数据量时）
```sql
-- 日志表按月分区
CREATE TABLE share_page_logs_2025_01 PARTITION OF share_page_logs
FOR VALUES FROM ('2025-01-01') TO ('2025-02-01');
```

### 2. 定期清理
```sql
-- 清理 90 天前的访问日志
DELETE FROM share_page_logs WHERE created_at < NOW() - INTERVAL '90 days';

-- 清理 365 天前的操作日志
DELETE FROM operation_logs WHERE created_at < NOW() - INTERVAL '365 days';
```

### 3. VACUUM 和 ANALYZE
```sql
-- 定期执行
VACUUM ANALYZE users;
VACUUM ANALYZE apple_accounts;
VACUUM ANALYZE unlock_tasks;
```
